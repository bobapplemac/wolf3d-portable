cmake_minimum_required(VERSION 3.16)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(STRINGS "${ROOT}/lib/wolf3d/VERSION" ENGINE_VERSION LIMIT_COUNT 1)
include("${ROOT}/cmake/W3PCheckEngineVersion.cmake")
w3p_check_engine_version("${ENGINE_VERSION}")
project(wolf3d-windows-modular VERSION ${ENGINE_VERSION} LANGUAGES C)
if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "The initial modular profile requires x86 MinGW")
endif()
set(WG_WINDOWS_BACKENDS "gdi,sdl1" CACHE STRING "Compiled Win2k backends: gdi,sdl1; gdi; or sdl1")
string(REPLACE "," ";" BACKENDS "${WG_WINDOWS_BACKENDS}")
foreach(backend IN LISTS BACKENDS)
 if(NOT backend STREQUAL "gdi" AND NOT backend STREQUAL "sdl1")
  message(FATAL_ERROR "Unknown backend: ${backend}")
 endif()
endforeach()
if(NOT BACKENDS)
 message(FATAL_ERROR "At least one backend is required")
endif()
list(REMOVE_DUPLICATES BACKENDS)
list(SORT BACKENDS)
string(REPLACE ";" "-" BACKEND_NAME "${BACKENDS}")
set(WG_DIST_CRT msvcrt CACHE STRING "Windows C runtime")
set(WG_DIST_PLATFORM win2k CACHE STRING "Windows baseline")
if(NOT WG_DIST_PLATFORM STREQUAL "win2k" OR NOT WG_DIST_CRT STREQUAL "msvcrt")
 message(FATAL_ERROR "The modular pilot targets Win2k/MSVCRT")
endif()
add_compile_definitions(_WIN32_WINNT=0x0500 WINVER=0x0500 WG_AUDIO_LEGACY_TARGET=1)
set(WG_BUILD_HEADLESS OFF CACHE BOOL "" FORCE)
option(WG_ENABLE_OPL_NUKED "Include Nuked" OFF)
option(WG_ENABLE_OPL_DBOPL "Include DBOPL" ON)
option(WG_ENABLE_OPL_ADLIB "Include native AdLib" ON)
set(WG_DEFAULT_OPL_DRIVER auto CACHE STRING "Default OPL driver")
add_subdirectory("${ROOT}/lib/wolf3d" wolf3d-lib)
function(configure_host target)
 target_sources(${target} PRIVATE "${ROOT}/platforms/WG_HELP.c" "${ROOT}/platforms/WG_TEXT_OUTPUT.c")
 target_link_libraries(${target} PRIVATE wolf3d user32 winmm)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
 target_link_options(${target} PRIVATE -static-libgcc -Wl,--major-subsystem-version,5,--minor-subsystem-version,0)
 if(WG_ENABLE_OPL_ADLIB)
  target_sources(${target} PRIVATE "${ROOT}/platforms/dos/WG_DOS_ADLIB.c" "${ROOT}/platforms/win32/WG_INPOUT32.c" "${ROOT}/platforms/win32/WG_OPL_PLAYBACK.c")
  target_compile_definitions(${target} PRIVATE WG_INPOUT32=1)
 endif()
endfunction()
add_executable(wolf3d-modular "${ROOT}/platforms/win32/WG_LAUNCHER.c")
configure_host(wolf3d-modular)
target_link_options(wolf3d-modular PRIVATE -municode)
set_target_properties(wolf3d-modular PROPERTIES OUTPUT_NAME wolf3d)
if("gdi" IN_LIST BACKENDS)
 target_sources(wolf3d-modular PRIVATE "${ROOT}/platforms/win32/WG_WIN32.c")
 target_compile_definitions(wolf3d-modular PRIVATE WG_HAVE_GDI=1 WG_BACKEND_EMBEDDED=1)
 target_link_libraries(wolf3d-modular PRIVATE gdi32 shell32)
endif()
set(SDL1_MODULE "")
if("sdl1" IN_LIST BACKENDS)
 set(SDL1_ROOT "" CACHE PATH "SDL1 prefix")
 find_path(SDL1_INCLUDE SDL/SDL.h HINTS "${SDL1_ROOT}/include" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
 find_library(SDL1_LIBRARY SDL HINTS "${SDL1_ROOT}/lib" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
 find_file(SDL1_DLL SDL.dll HINTS "${SDL1_ROOT}/bin" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
 find_file(SDL1_LICENSE COPYING HINTS "${SDL1_ROOT}" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
 add_library(wolf3d-sdl1-backend SHARED "${ROOT}/platforms/sdl1/WG_SDL1.c")
 configure_host(wolf3d-sdl1-backend)
 target_compile_definitions(wolf3d-sdl1-backend PRIVATE WG_BACKEND_MODULE=1)
 target_include_directories(wolf3d-sdl1-backend PRIVATE ${SDL1_INCLUDE})
 target_link_libraries(wolf3d-sdl1-backend PRIVATE ${SDL1_LIBRARY})
 set_target_properties(wolf3d-sdl1-backend PROPERTIES PREFIX "" OUTPUT_NAME wolf3d-sdl1)
 target_compile_definitions(wolf3d-modular PRIVATE WG_HAVE_SDL1=1)
 set(SDL1_MODULE "$<TARGET_FILE:wolf3d-sdl1-backend>")
 add_dependencies(wolf3d-modular wolf3d-sdl1-backend)
endif()
string(REGEX MATCH "^[0-9]+" COMPILER_MAJOR "${CMAKE_C_COMPILER_VERSION}")
set(DIST "${ROOT}/dist/wolf3d-portable_${PROJECT_VERSION}_win2k_x86_${BACKEND_NAME}_mingw-gcc${COMPILER_MAJOR}-msvcrt_modular")
add_custom_target(modular-release
 COMMAND ${CMAKE_COMMAND} "-DROOT=${ROOT}" "-DBUILD=${CMAKE_BINARY_DIR}" "-DDIST=${DIST}"
 "-DENGINE=$<TARGET_FILE:wolf3d>" "-DEXE=$<TARGET_FILE:wolf3d-modular>" "-DSDL=${SDL1_DLL}" "-DSDL_LICENSE=${SDL1_LICENSE}" "-DMODULE=${SDL1_MODULE}"
 "-DBACKENDS=${BACKEND_NAME}" "-DADLIB=${WG_ENABLE_OPL_ADLIB}" "-DNUKED=${WG_ENABLE_OPL_NUKED}" "-DDBOPL=${WG_ENABLE_OPL_DBOPL}"
 "-DVERSION=${PROJECT_VERSION}" "-DCOMPILER=${CMAKE_C_COMPILER_VERSION}"
 -P "${CMAKE_CURRENT_LIST_DIR}/package.cmake"
 DEPENDS wolf3d-modular VERBATIM)

include(CTest)
if(BUILD_TESTING)
 add_executable(wg-backend-test "${ROOT}/platforms/win32/WG_LAUNCHER.c" "${ROOT}/tests/WG_BACKEND_GDI_STUB.c" "${ROOT}/platforms/WG_HELP.c")
 target_compile_definitions(wg-backend-test PRIVATE WG_HAVE_GDI=1 WG_HAVE_SDL1=1)
 target_link_libraries(wg-backend-test PRIVATE wolf3d user32)
 target_link_options(wg-backend-test PRIVATE -municode -static-libgcc)
 target_compile_options(wg-backend-test PRIVATE -Wall -Wextra -Werror)
 foreach(kind good bad)
  add_library(wg-backend-${kind} SHARED "${ROOT}/tests/WG_BACKEND_MODULE_STUB.c")
  set_target_properties(wg-backend-${kind} PROPERTIES PREFIX "")
  target_link_options(wg-backend-${kind} PRIVATE -static-libgcc)
 endforeach()
 target_compile_definitions(wg-backend-bad PRIVATE TEST_ABI=999)
endif()
