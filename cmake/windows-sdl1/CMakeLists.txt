cmake_minimum_required(VERSION 3.16)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(STRINGS "${ROOT}/lib/wolf3d/VERSION" ENGINE_VERSION LIMIT_COUNT 1)
include("${ROOT}/cmake/W3PCheckEngineVersion.cmake")
w3p_check_engine_version("${ENGINE_VERSION}")
project(wolf3d-windows-sdl1 VERSION ${ENGINE_VERSION} LANGUAGES C)
if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "The initial SDL1 package profile requires x86 MinGW; legacy MSVC packaging is follow-up work")
endif()
set(WG_DIST_CRT msvcrt CACHE STRING "Windows C runtime")
set(WG_DIST_PLATFORM win2k CACHE STRING "Windows baseline")
if(NOT WG_DIST_PLATFORM STREQUAL "win2k" OR NOT WG_DIST_CRT STREQUAL "msvcrt")
 message(FATAL_ERROR "This initial packaging profile targets Win2k/MSVCRT")
endif()
add_compile_definitions(_WIN32_WINNT=0x0500 WINVER=0x0500)
set(WG_BUILD_HEADLESS OFF CACHE BOOL "" FORCE)
option(WG_ENABLE_OPL_NUKED "Include Nuked" OFF)
option(WG_ENABLE_OPL_DBOPL "Include DBOPL" ON)
option(WG_ENABLE_OPL_ADLIB "Include native AdLib" ON)
set(WG_DEFAULT_OPL_DRIVER auto CACHE STRING "Default OPL driver")
set(SDL1_ROOT "" CACHE PATH "SDL 1.2 installation prefix")
add_compile_definitions(WG_AUDIO_LEGACY_TARGET=1)
add_subdirectory("${ROOT}/lib/wolf3d" wolf3d-lib)
find_path(SDL1_INCLUDE SDL/SDL.h HINTS "${SDL1_ROOT}/include" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
find_library(SDL1_LIBRARY SDL HINTS "${SDL1_ROOT}/lib" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
find_file(SDL1_DLL SDL.dll HINTS "${SDL1_ROOT}/bin" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
find_file(SDL1_LICENSE COPYING HINTS "${SDL1_ROOT}" REQUIRED NO_CMAKE_FIND_ROOT_PATH)
add_executable(wolf3d-sdl1 "${ROOT}/platforms/sdl1/WG_SDL1.c" "${ROOT}/platforms/WG_HELP.c" "${ROOT}/platforms/WG_TEXT_OUTPUT.c")
target_include_directories(wolf3d-sdl1 PRIVATE ${SDL1_INCLUDE})
target_link_libraries(wolf3d-sdl1 PRIVATE wolf3d ${SDL1_LIBRARY} user32)
if(MINGW)
 target_compile_options(wolf3d-sdl1 PRIVATE -Wall -Wextra -Werror)
 target_link_options(wolf3d-sdl1 PRIVATE -static-libgcc -Wl,--major-subsystem-version,5,--minor-subsystem-version,0)
endif()
if(WG_ENABLE_OPL_ADLIB)
 target_sources(wolf3d-sdl1 PRIVATE "${ROOT}/platforms/dos/WG_DOS_ADLIB.c" "${ROOT}/platforms/win32/WG_INPOUT32.c" "${ROOT}/platforms/win32/WG_OPL_PLAYBACK.c")
 target_compile_definitions(wolf3d-sdl1 PRIVATE WG_INPOUT32=1)
 target_link_libraries(wolf3d-sdl1 PRIVATE winmm)
endif()
set_target_properties(wolf3d-sdl1 PROPERTIES OUTPUT_NAME wolf3d)
string(REGEX MATCH "^[0-9]+" COMPILER_MAJOR "${CMAKE_C_COMPILER_VERSION}")
set(DIST "${ROOT}/dist/wolf3d-portable_${PROJECT_VERSION}_win2k_x86_sdl1_mingw-gcc${COMPILER_MAJOR}-msvcrt")
add_custom_target(sdl1-release
 COMMAND ${CMAKE_COMMAND} "-DROOT=${ROOT}" "-DBUILD=${CMAKE_BINARY_DIR}" "-DDIST=${DIST}"
 "-DENGINE=$<TARGET_FILE:wolf3d>" "-DEXE=$<TARGET_FILE:wolf3d-sdl1>" "-DSDL=${SDL1_DLL}" "-DSDL_LICENSE=${SDL1_LICENSE}"
 "-DADLIB=${WG_ENABLE_OPL_ADLIB}" "-DNUKED=${WG_ENABLE_OPL_NUKED}" "-DDBOPL=${WG_ENABLE_OPL_DBOPL}"
 "-DVERSION=${PROJECT_VERSION}" "-DCOMPILER=${CMAKE_C_COMPILER_VERSION}"
 -P "${ROOT}/cmake/windows-sdl1/package.cmake"
 DEPENDS wolf3d-sdl1 VERBATIM)

include(CTest)
if(BUILD_TESTING)
 add_executable(wg-sdl1-test "${ROOT}/tests/WG_SDL1_TEST.c" "${ROOT}/platforms/WG_HELP.c" "${ROOT}/platforms/WG_TEXT_OUTPUT.c")
 target_include_directories(wg-sdl1-test PRIVATE ${SDL1_INCLUDE})
 target_link_libraries(wg-sdl1-test PRIVATE wolf3d ${SDL1_LIBRARY} user32)
 if(MINGW)
  target_link_options(wg-sdl1-test PRIVATE -static-libgcc)
 endif()
 add_test(NAME sdl1-host COMMAND wg-sdl1-test)
 set_tests_properties(sdl1-host PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
